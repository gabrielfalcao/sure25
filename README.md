# sure25 hashing algorithm


# Example
```
use sure25::Hasher;

let mut hasher = Hasher::new();
hasher.update(b"hello world");
assert_eq!(
    hasher.finalize(),
    vec![
        0xE8, 0x1C, 0xCA, 0x57, 0xF2, 0xD5, 0xDB, 0xCA, 0xEE, 0x19, 0x6A, 0xAC, 0xDD, 0x04,
        0xC3, 0x7E, 0x61, 0x62, 0x58, 0x84, 0x70, 0xDA, 0x70, 0xAF, 0x3F
    ]
);
```
